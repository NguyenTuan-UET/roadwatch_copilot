```python
"""Edge -> central event contract v1 (docs/report/trafficeye-edge-report.md, sections 6.6 and 13)."""

from typing import Literal
from uuid import UUID

from pydantic import AwareDatetime, BaseModel, Field

Direction = Literal["forward", "backward"]


class CountRow(BaseModel):
    vehicle_class: str = Field(min_length=1, max_length=32)
    direction: Direction
    count: int = Field(ge=1)


class CountPayload(BaseModel):
    line_id: str = Field(min_length=1, max_length=64)
    bucket_seconds: int = Field(gt=0, le=3600)
    rows: list[CountRow]  # empty list = no vehicle crossed the line during the bucket


class CountEvent(BaseModel):
    schema_version: Literal[1] = 1
    event_id: UUID
    seq: int = Field(ge=1)
    device_id: str = Field(min_length=1, max_length=64)
    camera_id: str = Field(min_length=1, max_length=64)
    type: Literal["count"] = "count"
    captured_at: AwareDatetime  # bucket start on the source clock (M-08)
    processed_at: AwareDatetime
    source: Literal["live", "replay"]
    payload: CountPayload


class EventBatch(BaseModel):
    events: list[CountEvent] = Field(min_length=1, max_length=500)


class BatchResult(BaseModel):
    accepted: int
    duplicates: int
```

```python
"""Plane geometry in image pixels (x to the right, y downward)."""

Point = tuple[float, float]


def side(a: Point, b: Point, p: Point) -> int:
    """Sign of the cross product (b - a) x (p - a): +1 or -1 by side of line a-b, 0 when p is on it."""
    cross = (b[0] - a[0]) * (p[1] - a[1]) - (b[1] - a[1]) * (p[0] - a[0])
    return (cross > 0) - (cross < 0)


def crosses_segment(p: Point, q: Point, a: Point, b: Point) -> bool:
    """Whether the move p -> q passes through segment a-b, given p and q lie strictly on opposite sides of line a-b."""
    return side(p, q, a) * side(p, q, b) <= 0
```

```python
"""Line-crossing counts grouped into time buckets (report section 8, steps 5 and 7)."""

from dataclasses import dataclass

from src.edge.geometry import Point, crosses_segment, side
from src.models.events import Direction

Box = tuple[float, float, float, float]


@dataclass(frozen=True)
class Track:
    track_id: int
    vehicle_class: str
    box: Box  # x1, y1, x2, y2 in pixels

    @property
    def bottom_center(self) -> Point:
        x1, _, x2, y2 = self.box
        return (x1 + x2) / 2, y2


class LineCounter:
    """Counts each track once, when its bottom-centre crosses segment a-b.

    "forward" = crossing from the right of a->b to its left as seen on the image. Draw the line left to right
    and a vehicle driving up the frame (away from a camera that films from behind) counts as forward.
    """

    def __init__(self, a: Point, b: Point):
        self.a, self.b = a, b
        self._last: dict[int, tuple[Point, int, int]] = {}  # track_id -> (last off-line point, its side, last frame)
        self._counted: set[int] = set()

    def update(self, track_id: int, point: Point, frame: int) -> Direction | None:
        prev = self._last.get(track_id)
        s = side(self.a, self.b, point)
        if s == 0:  # exactly on the line: decide once the track leaves it
            if prev:
                self._last[track_id] = (prev[0], prev[1], frame)
            return None
        self._last[track_id] = (point, s, frame)
        if prev is None or prev[1] == s or track_id in self._counted:
            return None
        if not crosses_segment(prev[0], point, self.a, self.b):
            return None
        self._counted.add(track_id)
        return "forward" if prev[1] > 0 else "backward"

    def prune(self, before_frame: int) -> None:
        """Forget tracks not seen since before_frame, so memory stays flat on a continuous stream."""
        for track_id in [t for t, (_, _, seen) in self._last.items() if seen < before_frame]:
            del self._last[track_id]
            self._counted.discard(track_id)
```

```python
class BucketCounts:
    """Groups crossings into fixed buckets of source time; a bucket is emitted once time moves past it."""

    def __init__(self, bucket_seconds: int = 60):
        self.bucket_seconds = bucket_seconds
        self._start: datetime | None = None
        self._counts: Counter[tuple[str, str]] = Counter()

    def tick(self, t: datetime) -> Bucket | None:
        """Call once per frame before add(); returns the bucket that just closed, if any."""
        start = datetime.fromtimestamp(int(t.timestamp()) // self.bucket_seconds * self.bucket_seconds, UTC)
        if start == self._start:
            return None
        closed = self.flush()
        self._start = start
        return closed

    def add(self, vehicle_class: str, direction: Direction) -> None:
        self._counts[(vehicle_class, direction)] += 1

    def flush(self) -> Bucket | None:
        if self._start is None:
            return None
        closed = (self._start, self._counts)
        self._start, self._counts = None, Counter()
        return closed


def uuid7() -> UUID:
    """RFC 9562 UUIDv7: 48-bit Unix-ms timestamp, then random bits (the stdlib has uuid7 only from Python 3.14)."""
    value = ((time.time_ns() // 1_000_000) << 80) | int.from_bytes(os.urandom(10), "big")
    value = (value & ~(0xF << 76)) | (0x7 << 76)  # version 7
    value = (value & ~(0x3 << 62)) | (0x2 << 62)  # RFC 4122 variant
    return UUID(int=value)


def iso(t: datetime) -> str:
    return t.astimezone(UTC).isoformat(timespec="milliseconds")


def make_count_event(
    *, device_id: str, camera_id: str, line_id: str, bucket: Bucket, bucket_seconds: int, source: str
) -> dict:
    """Event body for the outbox; the outbox adds `seq` when it stores the event (report section 13)."""
    start, counts = bucket
    return {
        "schema_version": 1,
        "event_id": str(uuid7()),
        "device_id": device_id,
        "camera_id": camera_id,
        "type": "count",
        "captured_at": iso(start),
        "processed_at": iso(datetime.now(UTC)),
        "source": source,
        "payload": {
            "line_id": line_id,
            "bucket_seconds": bucket_seconds,
            "rows": [{"vehicle_class": c, "direction": d, "count": n} for (c, d), n in sorted(counts.items())],
        },
    }
```

```python
"""Per-intersection scene config (report section 8, step 4). M1 needs only the counting line."""

from dataclasses import dataclass
from pathlib import Path

import yaml

from src.edge.geometry import Point


@dataclass(frozen=True)
class Scene:
    camera_id: str
    line_id: str
    line_a: Point
    line_b: Point
    classes: dict[int, str]  # detector class id -> name used in counts


def load_scene(path: Path) -> Scene:
    try:
        doc = yaml.safe_load(path.read_text(encoding="utf-8"))
        line = doc["count_line"]
        a, b = (float(line["a"][0]), float(line["a"][1])), (float(line["b"][0]), float(line["b"][1]))
        scene = Scene(str(doc["camera_id"]), str(line["id"]), a, b, {int(k): str(v) for k, v in doc["classes"].items()})
    except (KeyError, IndexError, TypeError, ValueError, AttributeError, yaml.YAMLError) as err:
        raise ValueError(f"{path}: invalid scene config ({err!r})") from err
    if a == b or not scene.classes:
        raise ValueError(f"{path}: count_line needs two different points and classes must not be empty")
    return scene
```

```python
from ultralytics import YOLO

from src.edge.counting import Track


class YoloTracker:
    def __init__(self, weights: str, classes: dict[int, str], imgsz: int = 640):
        self.model = YOLO(weights)  # downloads yolo26n.pt on first use
        self.classes = classes
        self.imgsz = imgsz

    def __call__(self, frame) -> list[Track]:
        boxes = self.model.track(
            frame,
            persist=True,  # keep ByteTrack state between frames of one stream
            tracker="bytetrack.yaml",
            classes=list(self.classes),
            imgsz=self.imgsz,
            verbose=False,
        )[0].boxes
        if boxes is None or boxes.id is None:  # no confirmed track in this frame
            return []
        return [
            Track(int(i), self.classes[int(c)], tuple(b))
            for i, c, b in zip(boxes.id.tolist(), boxes.cls.tolist(), boxes.xyxy.tolist(), strict=True)
        ]
```

```python
"""

import argparse
import logging
import os
import threading
import time
from datetime import UTC, datetime, timedelta
from pathlib import Path

import cv2

from src.edge.counting import Bucket, BucketCounts, LineCounter, make_count_event
from src.edge.outbox import Outbox
from src.edge.scene import load_scene
from src.edge.sender import http_post, run_forever
from src.edge.tracker import YoloTracker

BUCKET_S = 60
log = logging.getLogger("edge")


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="TrafficEye edge node (M1: counting).")
    parser.add_argument("--source", required=True, help="video file (15 FPS CFR) or rtsp:// URL")
    parser.add_argument("--scene", type=Path, required=True, help="scene YAML, e.g. configs/scenes/sample.yaml")
    parser.add_argument("--central", default=os.getenv("CENTRAL_URL", "http://localhost:8000"))
    parser.add_argument("--device-id", default=os.getenv("EDGE_DEVICE_ID", "edge-01"))
    parser.add_argument("--token", default=os.getenv("EDGE_TOKEN", ""), help="device token, see DEVICE_TOKENS")
    parser.add_argument("--weights", default="yolo26n.pt")
    parser.add_argument("--imgsz", type=int, default=640)
    parser.add_argument("--outbox", type=Path, default=Path("data/edge/outbox.sqlite"))
    parser.add_argument("--replay", action="store_true", help="mark events source=replay: simulated camera (M-08)")
    args = parser.parse_args(argv)
    logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(name)s: %(message)s")

    scene = load_scene(args.scene)
    from_file = not args.source.startswith("rtsp://")
    if not from_file:  # RTP over UDP does not pass Docker's TCP port mapping
        os.environ.setdefault("OPENCV_FFMPEG_CAPTURE_OPTIONS", "rtsp_transport;tcp")
    cap = cv2.VideoCapture(args.source)
    if not cap.isOpened():
        log.error("cannot open video source %s", args.source)
        return 2
    fps = cap.get(cv2.CAP_PROP_FPS)
    fps = fps if 1 <= fps <= 120 else 15.0  # some RTSP streams report 0 or 90000
    start = datetime.now(UTC).replace(second=0, microsecond=0)  # file second 0 = start of a minute bucket

    tracker = YoloTracker(args.weights, scene.classes, args.imgsz)
    line = LineCounter(scene.line_a, scene.line_b)
    buckets = BucketCounts(BUCKET_S)
    outbox = Outbox(args.outbox)
    stop = threading.Event()
    post = http_post(f"{args.central.rstrip('/')}/api/v1/events", args.token)
    sender = threading.Thread(target=run_forever, args=(args.outbox, post, stop), daemon=True)
    sender.start()

    def emit(bucket: Bucket) -> None:
        event = make_count_event(
            device_id=args.device_id,
            camera_id=scene.camera_id,
            line_id=scene.line_id,
            bucket=bucket,
            bucket_seconds=BUCKET_S,
            source="replay" if args.replay else "live",
        )
        seq = outbox.put(event)
        log.info("bucket %s: %s (seq %d)", event["captured_at"], dict(bucket[1]) or "no vehicles", seq)

    frame_idx, t0 = 0, time.monotonic()
    try:
        while True:
            ok, frame = cap.read()
            if not ok:  # end of file, or the stream dropped (Docker restart policy restarts the node)
                break
            t = start + timedelta(seconds=frame_idx / fps) if from_file else datetime.now(UTC)
            if closed := buckets.tick(t):
                emit(closed)
                line.prune(before_frame=frame_idx - int(30 * fps))
                log.info("%.1f FPS, %d events waiting in outbox", frame_idx / (time.monotonic() - t0), len(outbox))
            for track in tracker(frame):
                if direction := line.update(track.track_id, track.bottom_center, frame_idx):
                    buckets.add(track.vehicle_class, direction)
            frame_idx += 1
    except KeyboardInterrupt:
        log.info("interrupted")
    finally:
        cap.release()
        log.info("processed %d frames at %.1f FPS", frame_idx, frame_idx / max(time.monotonic() - t0, 1e-9))
        if closed := buckets.flush():
            emit(closed)
        deadline = time.monotonic() + 30  # give the sender a chance to deliver the last buckets
        while len(outbox) and time.monotonic() < deadline:
            time.sleep(0.5)
        stop.set()
        sender.join(timeout=5)
        log.info("stopped; %d events still queued in %s", len(outbox), args.outbox)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
"""Durable edge outbox: events wait in SQLite until central confirms them (report section 13)."""

import json
import sqlite3
from pathlib import Path


class Outbox:
    """Open one Outbox per thread: a sqlite3 connection must not be shared between threads."""

    def __init__(self, path: Path):
        path.parent.mkdir(parents=True, exist_ok=True)
        self._db = sqlite3.connect(path, timeout=10, isolation_level=None)  # autocommit: each write is on disk
        self._db.execute("PRAGMA journal_mode=WAL")
        # AUTOINCREMENT so a seq is never reused, even after the newest rows were acked and deleted
        self._db.execute(
            "CREATE TABLE IF NOT EXISTS outbox (seq INTEGER PRIMARY KEY AUTOINCREMENT, body TEXT NOT NULL)"
        )

    def put(self, event: dict) -> int:
        """Store an event (without seq); returns the seq it got."""
        return self._db.execute("INSERT INTO outbox (body) VALUES (?)", (json.dumps(event),)).lastrowid

    def peek(self, limit: int) -> list[dict]:
        rows = self._db.execute("SELECT seq, body FROM outbox ORDER BY seq LIMIT ?", (limit,)).fetchall()
        return [json.loads(body) | {"seq": seq} for seq, body in rows]

    def ack(self, up_to_seq: int) -> None:
        self._db.execute("DELETE FROM outbox WHERE seq <= ?", (up_to_seq,))

    def __len__(self) -> int:
        return self._db.execute("SELECT COUNT(*) FROM outbox").fetchone()[0]

    def close(self) -> None:
        self._db.close()
```

```python
"""Send outbox events to central over HTTP(S); delete them only after central confirms (design review H-01)."""

import json
import logging
import threading
import urllib.request
from collections.abc import Callable
from pathlib import Path

from src.edge.outbox import Outbox

log = logging.getLogger(__name__)

Post = Callable[[list[dict]], None]  # raises when central did not confirm


def http_post(url: str, token: str, timeout_s: float = 10.0) -> Post:
    def post(events: list[dict]) -> None:
        request = urllib.request.Request(
            url,
            data=json.dumps({"events": events}).encode(),
            headers={"Content-Type": "application/json", "Authorization": f"Bearer {token}"},
            method="POST",
        )
        with urllib.request.urlopen(request, timeout=timeout_s):  # raises HTTPError on 4xx/5xx
            pass

    return post


def drain(outbox: Outbox, post: Post, batch_size: int = 100) -> int:
    """Send everything queued, oldest first; returns how many events central confirmed."""
    sent = 0
    while batch := outbox.peek(batch_size):
        post(batch)
        outbox.ack(batch[-1]["seq"])
        sent += len(batch)
    return sent


def run_forever(
    outbox_path: Path, post: Post, stop: threading.Event, idle_s: float = 2.0, max_backoff_s: float = 30.0
) -> None:
    """Sender thread: drain, then wait; after a failure wait twice as long, up to max_backoff_s."""
    outbox = Outbox(outbox_path)  # opened inside the thread: sqlite3 objects must stay in the thread that made them
    delay = idle_s
    while not stop.is_set():
        try:
            if sent := drain(outbox, post):
                log.info("sent %d events to central", sent)
            delay = idle_s
        # ponytail: a batch central rejects (4xx) blocks the queue; add a dead-letter table if that ever happens
        except Exception as err:  # keep the thread alive; unconfirmed events stay queued, nothing is lost
            delay = min(delay * 2, max_backoff_s)
            log.warning("send failed (%s); %d events waiting; retry in %.0f s", err, len(outbox), delay)
        stop.wait(delay)
    outbox.close()
```

```python
"""Central event store. Plain SQL that runs on PostgreSQL (Docker Compose) and SQLite (tests, local runs)."""

import json
from collections import Counter
from datetime import UTC, datetime
from functools import lru_cache
from pathlib import Path

from sqlalchemy import Connection, Engine, create_engine, text

from src.config import get_settings
from src.models.events import CountEvent

SCHEMA = [
    """CREATE TABLE IF NOT EXISTS events (
        event_id TEXT PRIMARY KEY,
        device_id TEXT NOT NULL,
        camera_id TEXT NOT NULL,
        seq BIGINT NOT NULL,
        type TEXT NOT NULL,
        captured_at TEXT NOT NULL,
        processed_at TEXT NOT NULL,
        received_at TEXT NOT NULL,
        source TEXT NOT NULL,
        payload TEXT NOT NULL
    )""",
    "CREATE INDEX IF NOT EXISTS events_camera_type_time ON events (camera_id, type, captured_at)",
]

INSERT = text(
    "INSERT INTO events (event_id, device_id, camera_id, seq, type, captured_at, processed_at, received_at, source,"
    " payload) VALUES (:event_id, :device_id, :camera_id, :seq, :type, :captured_at, :processed_at, :received_at,"
    " :source, :payload) ON CONFLICT (event_id) DO NOTHING"
)


def iso(t: datetime) -> str:
    """UTC with fixed millisecond precision, so text timestamps sort in time order."""
    return t.astimezone(UTC).isoformat(timespec="milliseconds")


@lru_cache
def get_engine() -> Engine:
    url = get_settings().database_url
    if url.startswith("sqlite:///"):
        Path(url.removeprefix("sqlite:///")).parent.mkdir(parents=True, exist_ok=True)
    return create_engine(url)


def init_db(engine: Engine) -> None:
    with engine.begin() as conn:
        for statement in SCHEMA:
            conn.execute(text(statement))


def insert_events(conn: Connection, events: list[CountEvent]) -> int:
    """Store new events; a re-sent event (same event_id) is skipped. Returns how many were new."""
    received_at = iso(datetime.now(UTC))
    new = 0
    for e in events:
        params = {
            "event_id": str(e.event_id),
            "device_id": e.device_id,
            "camera_id": e.camera_id,
            "seq": e.seq,
            "type": e.type,
            "captured_at": iso(e.captured_at),
            "processed_at": iso(e.processed_at),
            "received_at": received_at,
            "source": e.source,
            "payload": e.payload.model_dump_json(),
        }
        new += conn.execute(INSERT, params).rowcount
    return new


def count_buckets(conn: Connection, camera_id: str, limit: int) -> list[dict]:
    """Newest count buckets of one camera, with totals per vehicle class and per direction."""
    rows = conn.execute(
        text(
            "SELECT captured_at, payload FROM events WHERE camera_id = :camera_id AND type = 'count'"
            " ORDER BY captured_at DESC LIMIT :limit"
        ),
        {"camera_id": camera_id, "limit": limit},
    ).all()
    buckets = []
    for captured_at, payload in rows:
        p = json.loads(payload)
        by_class: Counter[str] = Counter()
        by_direction: Counter[str] = Counter()
        for r in p["rows"]:
            by_class[r["vehicle_class"]] += r["count"]
            by_direction[r["direction"]] += r["count"]
        buckets.append(
            {
                "bucket_start": captured_at,
                "bucket_seconds": p["bucket_seconds"],
                "by_class": dict(by_class),
                "by_direction": dict(by_direction),
                "total": sum(by_class.values()),
            }
        )
    return buckets


def list_cameras(conn: Connection) -> list[dict]:
    rows = conn.execute(
        text("SELECT camera_id, MAX(received_at), COUNT(*) FROM events GROUP BY camera_id ORDER BY camera_id")
    ).all()
    return [{"camera_id": camera_id, "last_seen": seen, "events": n} for camera_id, seen, n in rows]
```

```python
"""Edge ingest and dashboard read endpoints (report sections 11 and 13)."""

import hmac

from fastapi import APIRouter, Depends, Header, HTTPException
from sqlalchemy import Engine

from src.config import Settings, get_settings
from src.db import get_engine, insert_events
from src.models.events import BatchResult, EventBatch

router = APIRouter()


def authenticated_device(
    authorization: str = Header(default=""),
    settings: Settings = Depends(get_settings),
) -> str:
    """Map the bearer token to its device_id. Compared as bytes in constant time."""
    scheme, _, presented = authorization.partition(" ")
    if scheme.lower() == "bearer" and presented:
        for pair in settings.device_tokens.split(","):
            device_id, _, token = pair.strip().partition(":")
            if token and hmac.compare_digest(token.encode(), presented.encode()):
                return device_id
    raise HTTPException(status_code=401, detail="invalid device token")


@router.post("/events", response_model=BatchResult)
def ingest_events(
    batch: EventBatch,
    device_id: str = Depends(authenticated_device),
    engine: Engine = Depends(get_engine),
) -> BatchResult:
    if any(e.device_id != device_id for e in batch.events):
        raise HTTPException(status_code=403, detail="device_id in events must match the token")
    with engine.begin() as conn:
        accepted = insert_events(conn, batch.events)
    return BatchResult(accepted=accepted, duplicates=len(batch.events) - accepted)
```

```ts
export type Camera = { camera_id: string; last_seen: string; events: number };

export type CountBucket = {
  bucket_start: string;
  bucket_seconds: number;
  by_class: Record<string, number>;
  by_direction: Record<string, number>;
  total: number;
};

const API_URL = process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:8000";

async function getJson<T>(path: string): Promise<T> {
  const res = await fetch(`${API_URL}/api/v1${path}`, { cache: "no-store" });
  if (!res.ok) throw new Error(`API ${res.status} at ${path}`);
  return (await res.json()) as T;
}

export async function fetchCameras(): Promise<Camera[]> {
  return (await getJson<{ cameras: Camera[] }>("/cameras")).cameras;
}

export async function fetchCounts(cameraId: string, limit = 30): Promise<CountBucket[]> {
  const query = `camera_id=${encodeURIComponent(cameraId)}&limit=${limit}`;
  return (await getJson<{ buckets: CountBucket[] }>(`/counts?${query}`)).buckets;
}
```

```python
#!/usr/bin/env python3
"""Add one recording session to an evaluation manifest (docs/eval/protocol.md, sections 2.6 and 8).

Videos must already be normalized to 15 FPS constant frame rate (design review H-02). Running the script again
for the same session replaces that session's rows.

Example:
  python scripts/make_manifest.py data/eval/dev/2026-09-24-sang --manifest eval/manifests/dev.csv \\
      --camera-id cam-01 --location C --condition day
"""

import argparse
import csv
import hashlib
import json
import subprocess
import sys
from fractions import Fraction
from pathlib import Path

FPS = Fraction(15)
COLUMNS = ["video_id", "session_id", "camera_id", "fps", "duration_s", "sha256", "location", "condition"]
NORMALIZE = (
    "ffmpeg -i IN.mp4 -vf \"scale=-2:'min(1080,ih)'\" -fps_mode cfr -r 15"
    " -c:v libx264 -preset veryfast -crf 23 -pix_fmt yuv420p -an OUT.mp4"
)


def probe(path: Path) -> tuple[Fraction, Fraction, float]:
    """(r_frame_rate, avg_frame_rate, duration_s) of the first video stream."""
    cmd = ["ffprobe", "-v", "error", "-select_streams", "v:0", "-show_entries", "stream=r_frame_rate,avg_frame_rate"]
    cmd += ["-show_entries", "format=duration", "-of", "json", str(path)]
    try:
        doc = json.loads(subprocess.run(cmd, capture_output=True, text=True, check=True).stdout)
    except FileNotFoundError as err:
        raise ValueError("ffprobe not found; install ffmpeg: conda install -c conda-forge ffmpeg") from err
    stream = doc["streams"][0]
    return Fraction(stream["r_frame_rate"]), Fraction(stream["avg_frame_rate"]), float(doc["format"]["duration"])


def sha256_file(path: Path) -> str:
    with path.open("rb") as f:
        return hashlib.file_digest(f, "sha256").hexdigest()


def session_rows(session_dir: Path, camera_id: str, location: str, condition: str) -> list[dict]:
    videos = sorted(session_dir.glob("*.mp4"))
    if not videos:
        raise ValueError(f"{session_dir}: no .mp4 files")
    rows = []
    for path in videos:
        r_rate, avg_rate, duration = probe(path)
        if r_rate != FPS or abs(avg_rate - FPS) > Fraction(1, 100):
            raise ValueError(f"{path}: {float(avg_rate):.3f} FPS, need constant 15 FPS. Normalize first: {NORMALIZE}")
        rows.append(
            {
                "video_id": path.stem,
                "session_id": session_dir.name,
                "camera_id": camera_id,
                "fps": 15,
                "duration_s": round(duration, 3),
                "sha256": sha256_file(path),
                "location": location,
                "condition": condition,
            }
        )
    return rows


def merge(manifest: Path, rows: list[dict]) -> list[dict]:
    """Replace this session's rows; refuse a video_id that another session already uses."""
    old: list[dict] = []
    if manifest.exists():
        with manifest.open(newline="", encoding="utf-8") as f:
            old = list(csv.DictReader(f))
    kept = [r for r in old if r["session_id"] != rows[0]["session_id"]]
    clash = {r["video_id"] for r in kept} & {r["video_id"] for r in rows}
    if clash:
        raise ValueError(f"{manifest}: video_id {sorted(clash)} already belongs to another session")
    return sorted(kept + rows, key=lambda r: r["video_id"])


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="Add one recording session to an evaluation manifest.")
    parser.add_argument("session_dir", type=Path, help="data/eval/<split>/<session_id>/")
    parser.add_argument("--manifest", type=Path, required=True, help="eval/manifests/<split>.csv")
    parser.add_argument("--camera-id", required=True)
    parser.add_argument("--location", required=True)
    parser.add_argument("--condition", required=True, help="day, night, rain, ...")
    args = parser.parse_args(argv)
    try:
        rows = merge(args.manifest, session_rows(args.session_dir, args.camera_id, args.location, args.condition))
    except (OSError, ValueError, subprocess.CalledProcessError) as err:
        print(f"error: {err}", file=sys.stderr)
        return 2
    args.manifest.parent.mkdir(parents=True, exist_ok=True)
    with args.manifest.open("w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=COLUMNS)
        writer.writeheader()
        writer.writerows(rows)
    print(f"{args.manifest}: {len(rows)} videos")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
#!/usr/bin/env python3
"""Measure decode + YOLO26 + ByteTrack FPS on a video file, the way the edge runs it (WP-31, later WP-23).

Example: python scripts/bench_fps.py data/samples/sample.mp4 --device 0
"""

import argparse
import time

import cv2
from ultralytics import YOLO

VEHICLE_CLASSES = [2, 3, 5, 7]  # COCO car, motorcycle, bus, truck (configs/scenes/*.yaml)


def main() -> None:
    parser = argparse.ArgumentParser(description="Measure detection + tracking FPS on a video file.")
    parser.add_argument("video")
    parser.add_argument("--weights", default="yolo26n.pt")
    parser.add_argument("--imgsz", type=int, default=640)
    parser.add_argument("--device", default="cpu", help="cpu, or 0 for the first NVIDIA GPU")
    args = parser.parse_args()

    model = YOLO(args.weights)
    options = {"persist": True, "tracker": "bytetrack.yaml", "classes": VEHICLE_CLASSES, "verbose": False}
    options |= {"imgsz": args.imgsz, "device": args.device}
    cap = cv2.VideoCapture(args.video)
    ok, frame = cap.read()
    if not ok:
        raise SystemExit(f"cannot read {args.video}")
    model.track(frame, **options)  # warm-up: model load and first inference are not counted
    frames, start = 0, time.perf_counter()
    while True:
        ok, frame = cap.read()
        if not ok:
            break
        model.track(frame, **options)
        frames += 1
    elapsed = time.perf_counter() - start
    print(f"{frames} frames in {elapsed:.1f} s = {frames / elapsed:.1f} FPS (device={args.device}, imgsz={args.imgsz})")


if __name__ == "__main__":
    main()
```
